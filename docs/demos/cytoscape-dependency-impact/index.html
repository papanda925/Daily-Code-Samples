実装コード
async v=>{const {default:cytoscape}=await lib('cytoscape@3.31.1');const host=panel();host.style.height='320px';const names={order:'注文',stock:'在庫',pay:'決済',gate:'外部ゲートウェイ'};const cy=cytoscape({container:host,elements:[...Object.keys(names).map(id=>({data:{id,label:names[id]}})),...[['order','stock'],['order','pay'],['pay','gate']].map(([source,target])=>({data:{source,target}}))],style:[{selector:'node',style:{label:'data(label)'}},{selector:'edge',style:{'target-arrow-shape':'triangle','curve-style':'bezier'}},{selector:'.active',style:{'background-color':'#dd653e','line-color':'#dd653e'}}],layout:{name:'breadthfirst',directed:true,padding:32}});let nodes=cy.$id('order');for(let i=0;i<(v==='2'?2:1);i++)nodes=nodes.union(nodes.outgoers());nodes.addClass('active');cleanup=()=>cy.destroy();say(nodes.nodes().map(n=>n.data('label')))}


結果の見方
1段では在庫と決済が対象。
2段で外部ゲートウェイに届く。
矢印は依存の方向を示す。
影響を受ける側から逆向きに探す場合は、先へ進む探索と区別する。


コードの要点
注文ノードの outgoers を強調する。


仕事で使うとき
障害影響の調査では、直接の依存先と間接の依存先で確認順を分ける。
